cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(reknot_jit_bridge LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
  message(FATAL_ERROR "The experimental bridge requires Windows x64 / MSVC")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(required LLVM_ROOT ZLIB_SOURCE_DIR ZSTD_SOURCE_DIR)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Set ${required}; see docs/JIT_CODE_OWNERSHIP.md")
  endif()
endforeach()
execute_process(COMMAND "${LLVM_ROOT}/bin/llvm-config.exe" --version
  OUTPUT_VARIABLE LLVM_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT LLVM_VERSION STREQUAL "23.1.0")
  message(FATAL_ERROR "This bridge and its local COFF patch require LLVM 23.1.0")
endif()
execute_process(COMMAND "${LLVM_ROOT}/bin/llvm-config.exe" --libfiles orcjit jitlink support
  OUTPUT_VARIABLE LLVM_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
separate_arguments(LLVM_LIBS WINDOWS_COMMAND "${LLVM_LIBS}")
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${ZLIB_SOURCE_DIR}" zlib-build EXCLUDE_FROM_ALL)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_MULTITHREAD_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${ZSTD_SOURCE_DIR}/build/cmake" zstd-build EXCLUDE_FROM_ALL)
execute_process(COMMAND "${LLVM_ROOT}/bin/llvm-tblgen.exe" -gen-opt-parser-defs
  -I "${LLVM_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llvm/COFFOptions.td"
  -o "${CMAKE_CURRENT_BINARY_DIR}/COFFOptions.inc" COMMAND_ERROR_IS_FATAL ANY)
add_library(reknot_jit_bridge SHARED bridge.cpp runtime.cpp vendor/llvm/COFFLinkGraphBuilder.cpp)
target_include_directories(reknot_jit_bridge PRIVATE "${LLVM_ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(reknot_jit_bridge PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(reknot_jit_bridge PRIVATE /O2 /GR- /utf-8 /Zc:preprocessor)
target_link_libraries(reknot_jit_bridge PRIVATE ${LLVM_LIBS} zlibstatic libzstd_static psapi shell32 ole32 uuid advapi32 ws2_32 ntdll)
option(REKNOT_JIT_TESTING "Export private fixture inspection helpers" OFF)
if(REKNOT_JIT_TESTING)
  target_compile_definitions(reknot_jit_bridge PRIVATE REKNOT_JIT_TESTING)
endif()
